# Find the largest number 
a = [1,2,33,43,54,45,654,78907,3453,22343]
largest = a[0]
for x in a:
  if x > a[0]:
   largest = x 
print(largest)
  

# Count how many times a specific element appears using a loop.

numbers = [1,2,3,3,3,4,5,5,5,6,6,6,6,7,8,787,89,786,786,0,0,2]
user = int(input("Enter a number:"))
i = 0
for x in numbers:
 if x == user:
   i += 1
print(user, "is", i, "times")
